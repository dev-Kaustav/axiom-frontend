export interface QuerySnapshot { /* as above */ isEmpty?: boolean }   // isEmpty: complete successful zero-item response
export type ViewState =
  | { kind: 'loading' }
  | { kind: 'empty' }
  | { kind: 'unavailable'; reason: 'not_configured' | 'capability' }
  | { kind: 'error'; error: ApiError }
  | { kind: 'rate_limited'; error: ApiError; retryAt: number }
  | { kind: 'stale_version'; error: ApiError }
